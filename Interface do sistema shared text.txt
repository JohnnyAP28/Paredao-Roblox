local function notif(texto, cor)
	local sg = LP.PlayerGui:FindFirstChild("OriginalNotif")
	if not sg then
		sg = Instance.new("ScreenGui")
		sg.Name = "OriginalNotif"
		sg.ResetOnSpawn = false
		sg.IgnoreGuiInset = true
		sg.DisplayOrder = 900
		sg.Parent = LP.PlayerGui
	end
	local lbl = Instance.new("TextLabel")
	lbl.Size = UDim2.new(0, 350, 0, 36)
	lbl.Position = UDim2.new(0.5, -175, 0, 16)
	lbl.BackgroundColor3 = cor or COR_ON
	lbl.BackgroundTransparency = 0.12
	lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
	lbl.Font = Enum.Font.GothamBold
	lbl.TextSize = 13
	lbl.TextWrapped = true
	lbl.Text = texto
	lbl.Parent = sg
	Instance.new("UICorner", lbl).CornerRadius = UDim.new(0, 8)
	task.delay(3.4, function()
		TweenService:Create(lbl, TweenInfo.new(0.4), {BackgroundTransparency = 1, TextTransparency = 1}):Play()
		task.wait(0.45)
		lbl:Destroy()
	end)
end

local REMOTES = {}
local function scanRemotes()
	REMOTES = {}
	for _, root in ipairs({ReplicatedStorage, workspace}) do
		pcall(function()
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
					table.insert(REMOTES, d)
				end
			end
		end)
	end
end
scanRemotes()

local function fireMatching(palavras, ...)
	local args = {...}
	local nomes = {}
	for _, r in ipairs(REMOTES) do
		local nm = string.lower(r.Name)
		for _, p in ipairs(palavras) do
			if string.find(nm, p, 1, true) then
				table.insert(nomes, r.Name)
				task.spawn(function()
					pcall(function()
						if r:IsA("RemoteEvent") then
							r:FireServer(table.unpack(args))
						else
							r:InvokeServer(table.unpack(args))
						end
					end)
				end)
				break
			end
		end
	end
	return nomes
end

local function flagValue(flag)
	local f = string.lower(flag)
	if string.find(f, "unlocked") then return true end
	if string.find(f, "locked")   then return false end
	if string.find(f, "price") or string.find(f, "cost") then return 0 end
	return true
end

local function applyFlags(obj)
	local achou = false
	for _, flag in ipairs(FLAGS) do
		if obj:GetAttribute(flag) ~= nil then
			pcall(function() obj:SetAttribute(flag, flagValue(flag)) end)
			achou = true
		end
		local v = obj:FindFirstChild(flag)
		if v and v:IsA("ValueBase") then
			pcall(function() v.Value = flagValue(flag) end)
			achou = true
		end
	end
	return achou
end

--#################### NOVO: TOQUE VIRTUAL E BUSCAS ####################--
local function estaVisivel(obj)
	local o = obj
	while o and o ~= LP.PlayerGui do
		if o:IsA("GuiObject") and not o.Visible then return false end
		if o:IsA("LayerCollector") and not o.Enabled then return false end
		o = o.Parent
	end
	return true
end

local function cliqueVirtual(botao)
	if not botao or not botao.Parent then return false, "botão inexistente" end
	local cam = workspace.CurrentCamera
	if not cam then return false, "sem câmera" end
	local pos = botao.AbsolutePosition + (botao.AbsoluteSize / 2)
	local ok, err = pcall(function()
		VirtualUser:Button1Down(Vector3.new(pos.X, pos.Y, 0), cam.CFrame)
		task.wait(0.07)
		VirtualUser:Button1Up(Vector3.new(pos.X, pos.Y, 0), cam.CFrame)
	end)
	if not ok then return false, tostring(err) end
	return true
end

local function acharBotoesPorTexto(fragmento, limite, exigirVisivel)
	local achados = {}
	fragmento = string.lower(fragmento)
	for _, g in ipairs(LP.PlayerGui:GetDescendants()) do
		if g:IsA("GuiButton") then
			local txt = g:IsA("TextButton") and string.lower(g.Text) or ""
			local alt = g:IsA("ImageButton") and string.lower(g.Name) or ""
			if string.find(txt, fragmento, 1, true) or string.find(alt, fragmento, 1, true) then
				if (not exigirVisivel) or (estaVisivel(g) and g.AbsoluteSize.X > 0) then
					table.insert(achados, g)
					if limite and #achados >= limite then break end
				end
			end
		end
	end
	return achados
end

local function liberarBotao(b)
	pcall(function()
		b.Active = true
		b.Visible = true
		if b.Interactable ~= nil then b.Interactable = true end
		if b.Selectable ~= nil then b.Selectable = true end
	end)
end

local function acharSons()
	local lista = {}
	for _, root in ipairs({workspace, SoundService, ReplicatedStorage}) do
		pcall(function()
			for _, d in ipairs(root:GetDescendants()) do
				if d:IsA("Sound") and d.IsPlaying and d.TimeLength > 0 then
					table.insert(lista, d)
				end
			end
		end)
	end
	table.sort(lista, function(a, b) return a.Volume > b.Volume end)
	return lista
end

--================================= VIP ================================--
local function aplicarVip(player)
	player = player or LP
	for _, n in ipairs(VIP_WORDS) do
		pcall(function() player:SetAttribute(n, true) end)
		pcall(function() player:SetAttribute(n:upper(), true) end)
	end
	local ls = player:FindFirstChild("leaderstats")
	if ls then
		for _, v in ipairs(ls:GetChildren()) do
			if v:IsA("ValueBase") then
				local nm = string.lower(v.Name)
				for _, w in ipairs(VIP_WORDS) do
					if string.find(nm, w, 1, true) then pcall(function() v.Value = true end) end
				end
			end
		end
	end
end

local function forcarAdmin(silencioso)
	for _, n in ipairs(ADMIN_WORDS) do
		pcall(function() LP:SetAttribute(n, true) end)
	end
	local botoes = {}
	pcall(function()
		for _, g in ipairs(LP.PlayerGui:GetDescendants()) do
			local nm = string.lower(g.Name)
			local bate = false
			for _, w in ipairs(ADMIN_WORDS) do
				if string.find(nm, w, 1, true) then bate = true break end
			end
			if bate then
				pcall(function()
					if g:IsA("ScreenGui") then
						g.Enabled = true
						g.DisplayOrder = 100
					elseif g:IsA("GuiObject") then
						g.Visible = true
						g.ZIndex = 50
						liberarBotao(g)
					end
				end)
				if g:IsA("GuiButton") then table.insert(botoes, g) end
			end
		end
	end)
	if not silencioso then
		notif(#botoes > 0 and ("Admin: " .. #botoes .. " botões destravados")
			or "Nenhum botão admin achado (use a lista)", #botoes > 0 and COR_OK or COR_OFF)
	end
	return botoes
end

local function abrirPainelAdmin()
	-- 1) tenta abrir os painéis direto
	local abertos = 0
	for _, p in ipairs(coletarPaineis()) do
		pcall(function()
			if p:IsA("ScreenGui") then p.Enabled = true
			else p.Visible = true end
			abertos += 1
		end)
	end
	-- 2) tenta simular o toque nos botões admin visíveis
	local tocados = 0
	for _, b in ipairs(acharBotoesPorTexto("admin", 6, true)) do
		liberarBotao(b)
		if cliqueVirtual(b) then tocados += 1 end
	end
	return abertos, tocados

--================================ SKINS ================================--
local skinsEncontradas = 0

local function coletarSkins()
	local lista, vistos = {}, {}
	for _, root in ipairs({ReplicatedStorage, LP.PlayerGui, LP:FindFirstChild("Backpack"), workspace}) do
		if root then
			pcall(function()
				for _, d in ipairs(root:GetDescendants()) do
					if not vistos[d] then
						local nm = string.lower(d.Name)
						local pai = d.Parent and string.lower(d.Parent.Name) or ""
						local bate = false
						for _, k in ipairs(SKIN_WORDS) do
							if string.find(nm, k, 1, true) or string.find(pai, k, 1, true) then bate = true break end
						end
						if bate then vistos[d] = true table.insert(lista, d) end
					end
				end
			end)
		end
	end
	return lista
end

local function idDoSkin(obj)
	return tostring(obj:GetAttribute("Id") or obj:GetAttribute("ID")
		or obj:GetAttribute("SkinId") or obj:GetAttribute("SkinID")
		or obj:GetAttribute("Name") or obj:GetAttribute("Nome") or obj.Name)
end

local function unlockSkins()
	local lista = coletarSkins()
	skinsEncontradas = 0
	local nomes = {}
	for _, obj in ipairs(lista) do
		if applyFlags(obj) then skinsEncontradas += 1 end
		for _, c in ipairs(obj:GetChildren()) do applyFlags(c) end
		if obj.Parent then applyFlags(obj.Parent) end
		table.insert(nomes, idDoSkin(obj))
		table.insert(nomes, obj.Name)
		for _, d in ipairs(obj:GetDescendants()) do
			if d:IsA("ProximityPrompt") then
				pcall(function() d.Enabled = true d.HoldDuration = 0 d.MaxActivationDistance = 1000 end)
			elseif d:IsA("ClickDetector") then
				pcall(function() d.MaxActivationDistance = 1000 end)
			end
		end
	end
	local unico, visto = {}, {}
	for _, n in ipairs(nomes) do
		if not visto[n] then visto[n] = true table.insert(unico, n) end
	end
	local texto = table.concat(unico, ",")
	pcall(function()
		for _, obj in ipairs({LP, LP:FindFirstChild("leaderstats"), LP:FindFirstChild("Skins"), LP:FindFirstChild("Data")}) do
			if obj then
				for _, a in ipairs(obj:GetAttributes()) do
					local an = string.lower(a)
					local val = obj:GetAttribute(a)
					if (string.find(an, "skin") or string.find(an, "own") or string.find(an, "unlock") or string.find(an, "invent")) then
						if type(val) == "table" and #val < #unico then obj:SetAttribute(a, unico)
						elseif type(val) == "string" and #val < #texto then obj:SetAttribute(a, texto) end
					end
				end
				for _, v in ipairs(obj:GetChildren()) do
					if v:IsA("StringValue") and (string.find(string.lower(v.Name), "skin") or string.find(string.lower(v.Name), "own")) then
						if #v.Value < #texto then v.Value = texto end
					end
				end
			end
		end
	end)
	return skinsEncontradas
end

local function comprarSkins()
	local alvos = {}
	for _, obj in ipairs(coletarSkins()) do
		local dono = obj:GetAttribute("Owned") or obj:GetAttribute("IsOwned")
			or obj:GetAttribute("Unlocked") or obj:GetAttribute("IsUnlocked")
		if dono ~= true then table.insert(alvos, obj) end
	end
	if #alvos == 0 then alvos = coletarSkins() end
	task.spawn(function()
		local n = 0
		for _, obj in ipairs(alvos) do
			n += 1
			if n > 90 then break end
			fireMatching({"buyskin","purchaseskin","unlockskin","comprarskin"}, obj.Name)
			task.wait(0.05)
			fireMatching({"buyskin","purchaseskin","unlockskin","equipskin","setskin"}, idDoSkin(obj))
			task.wait(0.05)
		end
		notif("Forçado: " .. n .. " skins via remotes", COR_ON)
	end)
end

local function injetarNoInventario()
	local inv
	for _, g in ipairs(LP.PlayerGui:GetDescendants()) do
		local nm = string.lower(g.Name)
		for _, w in ipairs(INV_WORDS) do
			if string.find(nm, w, 1, true) and (g:IsA("GuiObject") or g:IsA("ScreenGui")) then inv = g break end
		end
		if inv then break end
	end
	if not inv then return 0 end
	local grid
	for _, d in ipairs(inv:GetDescendants()) do
		if (d:IsA("UIGridLayout") or d:IsA("UIListLayout")) and d.Parent and #d.Parent:GetChildren() > 1 then
			grid = d.Parent break
		end
	end
	if not grid then return 0 end
	local template
	for _, c in ipairs(grid:GetChildren()) do
		if c:IsA("GuiObject") then template = c break end
	end
	if not template then return 0 end
	local existentes = {}
	for _, c in ipairs(grid:GetChildren()) do
		if c:IsA("GuiObject") then existentes[string.lower(c.Name)] = true end
	end
	local add = 0
	for _, obj in ipairs(coletarSkins()) do
		local nome = obj.Name
		if #nome <= 32 and not existentes[string.lower(nome)] then
			local ok, cl = pcall(function() return template:Clone() end)
			if ok and cl then
				cl.Name = nome
				local primeiro = true
				for _, d in ipairs(cl:GetDescendants()) do
					if d:IsA("ImageLabel") and string.find(string.lower(d.Name), "lock", 1, true) then d.Visible = false
					elseif d:IsA("TextLabel") and primeiro then d.Text = nome primeiro = false
					elseif d:IsA("ValueBase") then pcall(function() d.Value = flagValue(d.Name) end) end
				end
				cl.Visible = true
				cl.Parent = grid
				existentes[string.lower(nome)] = true
				add += 1
			end
		end
	end
	return add
end

--#################### NOVO: MÚSICA (DJ) ####################--
local function pularMusica()
	scanRemotes()
	local tocados = {}
	for _, w in ipairs(SKIP_WORDS) do
		local r = fireMatching({w}, true)
		for _, n in ipairs(r) do table.insert(tocados, n) end
	end
	-- tenta variações de argumento
	fireMatching({"skip","pular","pule"}, 0)
	fireMatching({"skip","pular","pule"}, "skip")

	-- reforço local: força o áudio atual a terminar
	local sons = acharSons()
	local locais = 0
	for i, s in ipairs(sons) do
		if i > 3 then break end
		pcall(function() s.TimePosition = math.max(0, s.TimeLength - 0.2) end)
		locais += 1
	end
	return #tocados, locais
end

local function solicitarMusica(nome)
	if not nome or nome == "" then return 0 end
	scanRemotes()
	local n = 0
	for _, w in ipairs(PLAY_WORDS) do
		n += #fireMatching({w}, nome)
	end
	return n
end

local function infoMusicaAtual()
	local sons = acharSons()
	if #sons == 0 then return "Nenhum áudio tocando" end
	local s = sons[1]
	local restante = math.max(0, s.TimeLength - s.TimePosition)
	return string.format("%s · %.0fs/%.0fs · vol %.2f", s.Name, s.TimePosition, s.TimeLength, s.Volume)
end

local function destravarLoja(silencioso)
	local marcados = 0
	for _, l in ipairs(LP.PlayerGui:GetDescendants()) do
		if l:IsA("TextLabel") or l:IsA("TextButton") then
			local t = string.lower(l.Text or "")
			if string.find(t, "comprar", 1, true) or string.find(t, "r$", 1, true) then
				pcall(function()
					if l:IsA("TextButton") then l.Text = "COMPRADO" else l.Text = "COMPRADO" end
					l.TextColor3 = Color3.fromRGB(255, 215, 90)
				end)
				marcados += 1
			end
		end
	end
	for _, b in ipairs(acharBotoesPorTexto("comprar", 40, false)) do liberarBotao(b) end
	aplicarVip(LP)
	scanRemotes()
	fireMatching(LOJA_WORDS)
	if not silencioso then
		notif("Loja: " .. marcados .. " itens marcados como COMPRADO", COR_OK)
	end
	return marcados
end

local function simularCompra()
	local alvos = acharBotoesPorTexto("comprar", 8, true)
	if #alvos == 0 then
		notif("Abra a Loja primeiro (nenhum COMPRAR visível)", COR_OFF)
		return 0
	end
	local n = 0
	for i, b in ipairs(alvos) do
		if i > 4 then break end
		liberarBotao(b)
		if cliqueVirtual(b) then n += 1 end
		task.wait(0.5)
	end
	notif("Toques simulados: " .. n, COR_ON)
	return n
end

local noclipConn

		local c = LP.Character

local flyBV, flyBG, flyConn
local function setFly(on)
	if flyConn then flyConn:Disconnect() flyConn = nil end
	if flyBV then flyBV:Destroy() flyBV = nil end
	if flyBG then flyBG:Destroy() flyBG = nil end
	if not on then return end
	local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
	if not root then return end
	flyBV = Instance.new("BodyVelocity")
	flyBV.MaxForce = Vector3.new(1e5, 1e5, 1e5)
	flyBV.Velocity = Vector3.zero
	flyBV.Parent = root
	flyBG = Instance.new("BodyGyro")
	flyBG.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
	flyBG.P = 1e4
	flyBG.Parent = root
	flyConn = RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		if not cam or not root.Parent then return end
		local move = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move -= Vector3.new(0, 1, 0) end
		flyBV.Velocity = move * CONFIG.FlySpeed
		flyBG.CFrame = cam.CFrame
		root.Velocity = Vector3.zero
	end)
end

	end
end

local ambBackup
local function setFullbright(on)
	if on then
		if not ambBackup then
			ambBackup = {Ambient = Lighting.Ambient, OutdoorAmbient = Lighting.OutdoorAmbient,
				Brightness = Lighting.Brightness, ClockTime = Lighting.ClockTime,
				FogEnd = Lighting.FogEnd, GlobalShadows = Lighting.GlobalShadows}
		end

local RELATORIO = ""

local function varrer(root, maxDepth, maxNodes, fn)
	local pilha = {{root, 0}}
	local cont = 0
	while #pilha > 0 do
		local item = table.remove(pilha)
		local obj, d = item[1], item[2]
		cont += 1
		if cont > maxNodes then return end
		local ok, parar = pcall(fn, obj, d)
		if not ok then parar = true end
		if not parar and d < maxDepth then
			for _, c in ipairs(obj:GetChildren()) do
				if not c:IsA("Terrain") then table.insert(pilha, {c, d + 1}) end
			end
		end
	end
end

local function attrsTxt(obj)
	local t = {}
	for _, a in ipairs(obj:GetAttributes()) do
		table.insert(t, a .. "=" .. tostring(obj:GetAttribute(a)))
	end
	return #t > 0 and (" [" .. table.concat(t, ", ") .. "]") or ""
end

--#################### NOVO: EXPORTAR / COPIAR (CORRIGIDO) ####################--
local function erroLimpo(e)
	return (tostring(e):gsub("\n", " "):sub(1, 110))
end

local function copiarRelatorio(resumido)
	local txt = gerarDiagnostico(resumido)
	RELATORIO = txt
	local erros = {}
	if setclipboard then
		local ok, err = pcall(setclipboard, txt)
		if ok then notif("Copiado (" .. #txt .. " chars). Cole onde quiser.", COR_OK) return true end
		table.insert(erros, "setclipboard: " .. erroLimpo(err))
	end
	if syn and syn.set_clipboard then
		local ok, err = pcall(syn.set_clipboard, txt)
		if ok then notif("Copiado via syn (" .. #txt .. " chars)", COR_OK) return true end
		table.insert(erros, "syn: " .. erroLimpo(err))
	end
	notif("Falha ao copiar -> " .. table.concat(erros, " | "), COR_ERR)
	return false
end

local function salvarRelatorio(resumido)
	local txt = gerarDiagnostico(resumido)
	RELATORIO = txt
	local nome = "diagnostico_original.txt"
	local erros, sucesso = {}, false

	local function tentar(fn, rotulo, nomeArq)
		if sucesso then return end
		local ok, err = pcall(fn, nomeArq, txt)
		if ok then sucesso = true else table.insert(erros, rotulo .. ": " .. erroLimpo(err)) end
	end

	if writefile then tentar(writefile, "writefile", nome) end
	if not sucesso and syn and syn.write_file then tentar(syn.write_file, "syn.write_file", nome) end
	if not sucesso and writefile then tentar(writefile, "writefile(/workspace)", "/workspace/" .. nome) end

	if sucesso then
		local dir = ""
		pcall(function() dir = getwritedir and getwritedir() or "" end)
		print("[original] arquivo salvo: " .. nome .. (dir ~= "" and (" em " .. dir) or ""))
		notif("Salvo: " .. nome .. " (pasta do executor)", COR_OK)
		return true
	end
	-- fallback: se não salvou, copia e avisa
	copiarRelatorio(resumido)
	notif("Sem writefile -> copiado. " .. table.concat(erros, " | "), COR_ERR)
	return false
end

local function exportarOnline()
	local txt = gerarDiagnostico(true)
	RELATORIO = txt
	if not request then notif("Executor sem 'request' (sem internet API)", COR_OFF) return end
	notif("Enviando relatório...", COR_ON)
	task.spawn(function()
		local ok, res = pcall(function()
			return request({
				Url = "https://paste.rs",
				Method = "PUT",
				Body = txt,
				Headers = {["Content-Type"] = "text/plain"},
			})
		end)
		if ok and res and res.Body then
			local link = tostring(res.Body):gsub("%s", "")
			RELATORIO = "LINK: " .. link .. "\n\n" .. txt
			pcall(function() setclipboard(link) end)
			notif("Link copiado: " .. link, COR_OK)
			print("[original] " .. link)
		else
			notif("Falha no envio: " .. erroLimpo(res), COR_ERR)
		end
	end)
end
