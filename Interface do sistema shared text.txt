local function flagValue(flag)
    local f = string.lower(flag)
    if string.find(f, "unlocked") then return true end
    if string.find(f, "locked")   then return false end
    if string.find(f, "price") or string.find(f, "cost") then return 0 end
    if string.find(f, "free") then return true end
    if string.find(f, "owned") or string.find(f, "bought") or string.find(f, "purchased") then return true end
    return nil
end

local REMOTES = {}
local function scanRemotes()
    REMOTES = {}
    for _, root in ipairs({RS, workspace}) do
        pcall(function()
            for _, d in ipairs(root:GetDescendants()) do
                if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
                    REMOTES[#REMOTES+1] = d
                end
            end
        end)
    end
    return #REMOTES
end

local function fireMatching(palavras, ...)
    local args = {...}; local usados = {}
    for _, r in ipairs(REMOTES) do
        local nm = string.lower(r.Name)
        for _, p in ipairs(palavras) do
            if string.find(nm, p, 1, true) then
                usados[#usados+1] = r.Name
                task.spawn(function()
                    pcall(function()
                        if r:IsA("RemoteEvent") then r:FireServer(table.unpack(args))
                        else r:InvokeServer(table.unpack(args)) end
                    end)
                end)
                break
            end
        end
    end
    return usados
end

local function acharRemotes(palavras)
    local achados = {}
    for _, r in ipairs(REMOTES) do
        local nm = string.lower(r.Name)
        for _, p in ipairs(palavras) do
            if string.find(nm, p, 1, true) then achados[#achados+1] = r.Name; break end
        end
    end
    return achados
end
scanRemotes()

local function sendChat(msg)
    local ok = pcall(function()
        local ev = RS:FindFirstChild("EnviarMensagemEvent")
        if ev and ev:IsA("RemoteEvent") then ev:FireServer(msg); return end
        local ce = RS:FindFirstChild("DefaultChatSystemChatEvents")
        if ce and ce:FindFirstChild("SayMessageRequest") then
            ce.SayMessageRequest:FireServer(msg, "All"); return
        end
        error("sem remote")
    end)
    if not ok then pcall(function() StarterGui:SetCore("ChatSendMessage", msg) end) end
end

    DS.Estado, DS.Filtro, DS.Busca, DS.RenderMax = "parado", "TODOS", "", 200
    DS.Cfg = {depth=8, maxObj=25000, lote=320}
    DS._cancel, DS._pause = false, false
    DS.ui = nil

    local CLASSES = {
        ["Script"]=true, ["LocalScript"]=true, ["ModuleScript"]=true,
        ["Folder"]=true, ["Model"]=true, ["Tool"]=true, ["Part"]=true, ["MeshPart"]=true,
        ["Sound"]=true, ["ParticleEmitter"]=true, ["ScreenGui"]=true,
    }
    local function classeGrupo(c)
        if c == "Script" or c == "LocalScript" or c == "ModuleScript" then return "Scripts" end
        if c == "Folder" then return "Pastas" end
        if c == "Model" then return "Modelos" end
        if c == "Tool" then return "Ferramentas" end
        if c:find("Particle") or c:find("Trail") or c:find("Beam") or c:find("Emitter") then return "Efeitos" end
        if c:find("Sound") then return "Sons" end
        if c:find("Gui") or c:find("Frame") or c:find("Button")
           or c:find("Label") or c:find("TextBox") or c:find("Image") then return "Interfaces" end
        if c:find("Part") or c:find("Wedge") or c:find("Mesh") or c:find("Union") then return "Peças" end
        return "Outros"
    end

    function DS.log(msg, nivel)
        nivel = nivel or "INFO"
        DS.Logs[#DS.Logs+1] = {t=os.date("%H:%M:%S"), m=msg, n=nivel}
        if DS.ui and DS.ui.logs then
            local l = Instance.new("TextLabel")
            l.Size = UDim2.new(1,0,0,15); l.BackgroundTransparency = 1
            l.Font = Enum.Font.Code; l.TextSize = 11
            l.TextXAlignment = Enum.TextXAlignment.Left
            l.Text = "["..DS.Logs[#DS.Logs].t.."]["..nivel.."] "..msg
            l.TextColor3 = nivel=="ERRO" and Color3.fromRGB(255,120,120)
                or nivel=="AVISO" and Color3.fromRGB(255,210,120) or Color3.fromRGB(190,190,200)
            l.Parent = DS.ui.logs
        end
        if #DS.Logs > 500 then table.remove(DS.Logs, 1) end
    end

    function DS.detectarCaps()
        local c = DS.Caps
        c.getgenv = type(getgenv) == "function"
        c.gethui = type(gethui) == "function"
        c.setclipboard = type(setclipboard) == "function"
        c.toclipboard = type(toclipboard) == "function"
        c.writefile = type(writefile) == "function"
        c.readfile = type(readfile) == "function"
        c.isfile = type(isfile) == "function"
        c.listfiles = type(listfiles) == "function"
        c.makefolder = type(makefolder) == "function"
        c.getscriptbytecode = type(getscriptbytecode) == "function"
        c.getloadedmodules = type(getloadedmodules) == "function"
        c.gethiddenproperty = type(gethiddenproperty) == "function"
        c.getinstances = type(getinstances) == "function"
        c.identifyexecutor = type(identifyexecutor) == "function"
        if c.identifyexecutor then
            local ok, nome = pcall(identifyexecutor)
            c.executor = (ok and nome) and tostring(nome) or "desconhecido"
        else
            c.executor = "desconhecido"
        end
        c.escrita = c.writefile
        c.clipboard = c.setclipboard or c.toclipboard
        return c
    end

    function DS.fonte(sc)
        if type(getscriptbytecode) == "function" then
            local ok, bc = pcall(getscriptbytecode, sc)
            if ok and bc then
                local n = #tostring(bc)
                if n > 0 then return true, "bytecode disponível ("..n.." bytes)" end
            end
        end
        return false, "fonte não verificável (sem getscriptbytecode)"
    end

    local function registrar(inst, caminho, prof)
        local e = {nome=inst.Name, classe=inst.ClassName, caminho=caminho, prof=prof, grupo=classeGrupo(inst.ClassName)}
        if CLASSES[inst.ClassName] then
            local k = DS.Stats[e.classe]
            if k then DS.Stats[k] = DS.Stats[k] + 1 end
            if e.grupo == "Scripts" then
                local ok, motivo = DS.fonte(inst)
                e.fonteOk, e.fonteMotivo = ok, motivo
            end
        end
        DS.Stats.total = DS.Stats.total + 1
        DS.Results[#DS.Results+1] = e
    end

    function DS.varrerSubarvore(raiz, profMax)
        profMax = profMax or DS.Cfg.depth
        local fila = {{inst=raiz, prof=0, cam=(raiz==game and "game" or raiz.Name)}}
        local h = 1
        while h <= #fila do
            local it = fila[h]; h = h + 1
            local ok, filhos = pcall(function() return it.inst:GetChildren() end)
            if ok then
                for _, f in ipairs(filhos) do
                    local cam = it.cam.."."..f.Name
                    registrar(f, cam, it.prof+1)
                    if it.prof+1 < profMax then fila[#fila+1] = {inst=f, prof=it.prof+1, cam=cam} end
                end
            else
                DS.Stats.Erros = DS.Stats.Erros + 1
            end
        end
        return DS.Stats.total
    end

    local function zerarStats()
        DS.Stats = {total=0, Script=0, LocalScript=0, ModuleScript=0, Folder=0, Model=0,
                    Gui=0, Tool=0, Part=0, Sound=0, Erros=0}
    end

    function DS.limpar()
        DS.Results, DS.Filtrados = {}, {}
        zerarStats(); DS.Estado = "parado"
        DS.render(); DS.atualizarHUD()
        DS.log("Resultados da sessão limpos")
    end

    function DS.limparLogs()
        DS.Logs = {}
        if DS.ui and DS.ui.logs then
            for _, c in ipairs(DS.ui.logs:GetChildren()) do
                if not c:IsA("UIListLayout") then c:Destroy() end
            end
        end
    end

    function DS.iniciar()
        if DS.Estado == "rodando" or DS.Estado == "pausado" then
            DS.log("Varredura já em andamento", "AVISO"); return
        end
        DS.Results, DS.Filtrados = {}, {}
        zerarStats()
        DS._cancel, DS._pause, DS.Estado = false, false, "rodando"
        DS.log("Varredura iniciada (prof "..DS.Cfg.depth..", máx "..DS.Cfg.maxObj.." objetos)")
        DS.atualizarHUD()
        task.spawn(function()
            local fila = {{inst=game, prof=0, cam="game"}}
            local h = 1
            while h <= #fila do
                if DS._cancel then break end
                while DS._pause and not DS._cancel do task.wait(0.1) end
                local it = fila[h]; h = h + 1
                local ok, filhos = pcall(function() return it.inst:GetChildren() end)
                if ok then
                    for _, f in ipairs(filhos) do
                        local cam = it.cam.."."..f.Name
                        registrar(f, cam, it.prof+1)
                        if it.prof+1 < DS.Cfg.depth and DS.Stats.total < DS.Cfg.maxObj then
                            fila[#fila+1] = {inst=f, prof=it.prof+1, cam=cam}
                        end
                    end
                else
                    DS.Stats.Erros = DS.Stats.Erros + 1
                end
                if h % DS.Cfg.lote == 0 then
                    DS.atualizarHUD(); DS.filtrar(true); task.wait()
                end
                if DS.Stats.total >= DS.Cfg.maxObj then break end
            end
            DS.Estado = DS._cancel and "cancelado" or "concluido"
            DS.log("Varredura "..(DS._cancel and "CANCELADA" or "concluída").." · "..DS.Stats.total.." objetos",
                   DS._cancel and "AVISO" or "INFO")
            DS.atualizarHUD(); DS.filtrar(true)
        end)
    end

    function DS.pausar()
        if DS.Estado ~= "rodando" and DS.Estado ~= "pausado" then return end
        DS._pause = not DS._pause
        DS.Estado = DS._pause and "pausado" or "rodando"
        DS.log(DS._pause and "Pausado" or "Continuado", "AVISO")
        DS.atualizarHUD()
    end

    function DS.cancelar()
        if DS.Estado == "parado" then return end
        DS._cancel = true; DS._pause = false
        DS.log("Cancelamento solicitado", "AVISO")
    end

    function DS.filtrar(soRender)
        local base = (DS.Filtro == "IMPORTADOS") and DS.Importados or DS.Results
        local q = string.lower(DS.Busca)
        local out = {}
        for _, e in ipairs(base) do
            local okF = DS.Filtro == "TODOS" or DS.Filtro == "IMPORTADOS"
                        or e.grupo == DS.Filtro or e.classe == DS.Filtro
            local okB = q == "" or string.find(string.lower(e.nome), q, 1, true)
                        or string.find(string.lower(e.caminho), q, 1, true)
                        or string.find(string.lower(e.classe), q, 1, true)
            if okF and okB then out[#out+1] = e end
        end
        DS.Filtrados = out
        if not soRender then DS.render() end
    end

    function DS.relatorio()
        local l = {}
        l[#l+1] = "DEEPSCAN MOBILE — RELATÓRIO"
        l[#l+1] = "Data: "..os.date("%d/%m/%Y %H:%M:%S")
        l[#l+1] = "Executor: "..tostring(DS.Caps.executor).." | Estado: "..DS.Estado
        l[#l+1] = "Objetos: "..DS.Stats.total.." | Scripts: "..(DS.Stats.Script+DS.Stats.LocalScript+DS.Stats.ModuleScript)
        l[#l+1] = "Erros de leitura: "..DS.Stats.Erros
        l[#l+1] = string.rep("-", 46)
        for _, e in ipairs(DS.Results) do
            l[#l+1] = string.rep("  ", e.prof)..e.nome.." ["..e.classe.."] :: "..e.caminho
            if e.fonteMotivo then l[#l+1] = string.rep("  ", e.prof+1).."> "..e.fonteMotivo end
        end
        return table.concat(l, "\n")
    end

    function DS.exportarJSON()
        local dados = {data=os.date("%Y-%m-%d %H:%M:%S"), executor=DS.Caps.executor,
                       stats=DS.Stats, total=#DS.Results, objetos={}}
        for i, e in ipairs(DS.Results) do
            dados.objetos[i] = {nome=e.nome, classe=e.classe, caminho=e.caminho,
                                prof=e.prof, grupo=e.grupo, fonte=e.fonteMotivo}
        end
        local ok, js = pcall(function() return Http:JSONEncode(dados) end)
        if not ok then return nil, "falha ao codificar JSON" end
        local ok2, dec = pcall(function() return Http:JSONDecode(js) end)
        if not ok2 or type(dec) ~= "table" then return nil, "JSON gerado inválido" end
        return js
    end

    function DS.exportarTXT() return DS.relatorio() end

    function DS.copiar(txt)
        txt = txt or DS.relatorio()
        if setClip(txt) then return true, "copiado" end
        return false, "sem clipboard — use o painel de logs"
    end

    function DS.baixar(formato)
        local txt = (formato == "JSON") and DS.exportarJSON() or DS.exportarTXT()
        if not txt then return false, "falha ao gerar" end
        local nome = "DeepScan_"..os.date("%Y%m%d_%H%M%S")..((formato=="JSON") and ".json" or ".txt")
        if type(writefile) == "function" then
            local ok = pcall(writefile, nome, txt)
            if ok then
                local conf = "não confirmado"
                if type(isfile) == "function" then
                    local okf, existe = pcall(isfile, nome)
                    if okf and existe then conf = "confirmado" end
                end
                DS.log("Arquivo salvo: "..nome.." ("..conf..")")
                return true, "salvo em "..nome.." ("..conf..")"
            end
            DS.log("writefile falhou", "ERRO")
        end
        if setClip(txt) then
            DS.log("Sem writefile — conteúdo enviado ao clipboard", "AVISO")
            return true, "sem writefile: copiado para o clipboard"
        end
        return false, "sem gravação disponível"
    end

    function DS.importarTexto(txt)
        if not txt or txt == "" then return false, "texto vazio" end
        local ok, dados = pcall(function() return Http:JSONDecode(txt) end)
        if not ok or type(dados) ~= "table" or type(dados.objetos) ~= "table" then
            return false, "JSON inválido ou fora do formato DeepScan"
        end
        DS.Importados = dados.objetos
        DS.log("Importados "..#dados.objetos.." objetos de relatório externo")
        DS.Filtro = "IMPORTADOS"; DS.filtrar()
        return true, "importado: "..#dados.objetos.." objetos"
    end

    function DS.importarDoCampo(txt)
        if not txt or txt == "" then return false, "campo vazio" end
        local t = txt:gsub("^%s+", ""):gsub("%s+$", "")
        if t:sub(1, 1) ~= "{" then
            if type(readfile) == "function" then
                local ok, conteudo = pcall(readfile, t)
                if ok and conteudo then return DS.importarTexto(conteudo) end
            end
            return false, "não é JSON nem arquivo legível"
        end
        return DS.importarTexto(t)
    end

    function DS.visualizar(e)
        if not e or not DS.ui then return end
        local p = DS.ui.detalhe
        for _, c in ipairs(p:GetChildren()) do
            if not c:IsA("UIListLayout") then c:Destroy() end
        end
        local linhas = {
            "Nome: "..e.nome,
            "Classe: "..e.classe,
            "Caminho: "..e.caminho,
            "Profundidade: "..tostring(e.prof),
            "Grupo: "..e.grupo,
            "Fonte: "..(e.fonteMotivo or "não é script"),
        }
        for i, t in ipairs(linhas) do
            local l = Instance.new("TextLabel")
            l.Size = UDim2.new(1,-4,0,15); l.Position = UDim2.new(0,2,0,(i-1)*15)
            l.BackgroundTransparency = 1; l.Font = Enum.Font.Code; l.TextSize = 11
            l.TextXAlignment = Enum.TextXAlignment.Left; l.Text = t
            l.TextTruncate = Enum.TextTruncate.AtEnd
            l.TextColor3 = Color3.fromRGB(215,215,225); l.Parent = p
        end
    end
end

            l.Size = UDim2.new(1,0,0,16); l.BackgroundColor3 = Color3.fromRGB(24,23,36)
            l.Text = string.rep(" ", math.min(e.prof or 0, 10))..e.nome.."  ["..e.classe.."]"
            l.TextColor3 = e.fonteOk == false and Color3.fromRGB(255,180,120) or Color3.fromRGB(205,205,220)
            l.Font = Enum.Font.Code; l.TextSize = 11; l.TextTruncate = Enum.TextTruncate.AtEnd
            l.TextXAlignment = Enum.TextXAlignment.Left; l.AutoButtonColor = false; l.Parent = DS.ui.res
            l.MouseButton1Click:Connect(function() DS.visualizar(e) end)
        end
        DS._renderizado = alvo
    end

    res:GetPropertyChangedSignal("CanvasPosition"):Connect(function()
        if res.CanvasPosition.Y + res.AbsoluteWindowSize.Y >= res.AbsoluteCanvasSize.Y - 40 then
            if DS._renderizado < #DS.Filtrados then DS.adicionarLote() end
        end
    end)
end

function DeepScan.abrir()
    local ok, err = pcall(function()
        DeepScan.detectarCaps()
        DeepScan.construirUI()
        DeepScan.ui.gui.Enabled = true
        DeepScan.atualizarHUD()
        DeepScan.log("Interface aberta · executor: "..tostring(DeepScan.Caps.executor))
    end)
    if not ok then return false, tostring(err) end
    return true
end

--======================= 1.7 TESTES ==========================--
function DeepScan.testes()
    local DS = DeepScan
    if not DS.ui then
        local ok, err = DS.abrir()
        if not ok then notif("DeepScan: "..tostring(err), COR_ERR); return end
    end
    DS.log("==== INÍCIO DOS TESTES ====", "AVISO")
    local backupR, backupS, backupF, backupE = DS.Results, DS.Stats, DS.Filtrados, DS.Estado
    local function T(nome, fn)
        local ok, res = pcall(fn)
        local passou = ok and (res == true or type(res) == "string")
        DS.log(("TESTE %-38s → %s"):format(nome, passou and ("PASSOU ("..tostring(res)..")") or ("FALHOU: "..tostring(res))),
               passou and "INFO" or "ERRO")
    end
    local pasta = Instance.new("Folder"); pasta.Name = "DeepScanTeste"; pasta.Parent = workspace
    local s3 = Instance.new("ModuleScript"); s3.Name = "TesteModulo"; s3.Parent = pasta
    local s1 = Instance.new("Script"); s1.Name = "TesteScript"; s1.Parent = pasta
    local s2 = Instance.new("LocalScript"); s2.Name = "TesteLocal"; s2.Parent = pasta

    T("1 inicialização", function()
        assert(DS.ui and DS.ui.gui, "UI ausente")
        local c = DS.detectarCaps(); local n = 0; for _ in pairs(c) do n = n + 1 end
        return n.." capacidades detectadas"
    end)
    T("2 varredura de subárvore", function()
        DS.Results = {}
        DS.Stats = {total=0, Script=0, LocalScript=0, ModuleScript=0, Folder=0, Model=0, Gui=0, Tool=0, Part=0, Sound=0, Erros=0}
        local n = DS.varrerSubarvore(pasta, 3)
        assert(n >= 3, "esperado >=3, obtido "..n)
        return n.." objetos reais"
    end)
    T("3 contagem por classe", function()
        assert(DS.Stats.Script == 1 and DS.Stats.LocalScript == 1 and DS.Stats.ModuleScript == 1, "contadores errados")
        return "Script=1 LocalScript=1 ModuleScript=1"
    end)
    T("4 filtro por classe", function()
        DS.Filtro = "Scripts"; DS.Busca = ""; DS.filtrar(true)
        for _, e in ipairs(DS.Filtrados) do assert(e.grupo == "Scripts", "vazou: "..e.classe) end
        local n = #DS.Filtrados
        DS.Filtro = "TODOS"; DS.filtrar(true)
        return n.." scripts filtrados"
    end)
    T("5 cópia", function()
        local ok, msg = DS.copiar("DeepScan teste")
        return ok and "clipboard ok" or ("fallback: "..msg)
    end)
    T("6 exportação JSON", function()
        local js = DS.exportarJSON(); assert(js, "json nulo")
        local d = Http:JSONDecode(js)
        assert(type(d) == "table" and type(d.objetos) == "table", "estrutura inválida")
        return #js.." bytes válidos"
    end)
    T("7 exportação TXT", function()
        local t = DS.exportarTXT()
        assert(type(t) == "string" and #t > 50 and string.find(t, "DEEPSCAN"), "txt inválido")
        return #t.." bytes"
    end)
    T("8 importar JSON válido", function()
        local ok, msg = DS.importarTexto(DS.exportarJSON()); assert(ok, msg); return msg
    end)
    T("9 rejeitar JSON inválido", function()
        assert(not DS.importarTexto("{isso nao e json"), "aceitou inválido"); return "rejeitado"
    end)
    T("10 gravação de arquivo", function()
        if type(writefile) == "function" then
            local ok, msg = DS.baixar("TXT"); assert(ok, msg); return "gravado"
        end
        return "sem writefile — fallback informado"
    end)
    T("11 pausa e continuação", function()
        DS.Estado = "rodando"; DS._pause = false
        DS.pausar(); assert(DS._pause and DS.Estado == "pausado", "não pausou")
        DS.pausar(); assert(not DS._pause and DS.Estado == "rodando", "não continuou")
        return "ok"
    end)
    T("12 cancelamento", function()
        DS._cancel = false; DS.Estado = "rodando"; DS.cancelar()
        assert(DS._cancel, "flag não marcada"); DS._cancel = false; return "flag ok"
    end)
    T("13 preservação de parciais", function()
        assert(#DS.Results > 0, "resultados perdidos"); return #DS.Results.." preservados"
    end)
    T("14 fonte não acessível", function()
        local ok, motivo = DS.fonte(s3)
        return ok and ("legível: "..motivo) or ("correto: "..motivo)
    end)

    pasta:Destroy()
    if #backupR > 0 then
        DS.Results, DS.Stats, DS.Filtrados = backupR, backupS, backupF
    else
        DS.Filtrados = DS.Results
    end
    DS.Estado = "parado"; DS.Filtro = "TODOS"
    DS.render(); DS.atualizarHUD()
    DS.log("==== FIM DOS TESTES ====", "AVISO")
    notif("DeepScan: testes concluídos (veja os logs)", COR_ON)
end
