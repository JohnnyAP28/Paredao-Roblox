do
    Util.IsMobile = UIS.TouchEnabled and not UIS.KeyboardEnabled

    function Util.HasDrawing()
        return type(rawget(getfenv(0), "Drawing")) == "table"
            or (typeof and typeof(Drawing) == "table")
    end

    function Util.FileAPI()
        return type(writefile) == "function"
           and type(readfile) == "function"
           and type(isfile) == "function"
    end

    function Util.Clipboard(txt)
        if type(setclipboard) == "function" then setclipboard(txt)
        elseif type(toclipboard) == "function" then toclipboard(txt) end
    end

    function Util.Clamp(v, a, b) return math.max(a, math.min(b, v)) end
    function Util.Round(v, d)
        local m = 10 ^ (d or 0)
        return math.floor(v * m + 0.5) / m
    end

    function Util.Format(n)
        n = tonumber(n) or 0
        local units = { {1e12,"T"}, {1e9,"B"}, {1e6,"M"}, {1e3,"K"} }
        for _, u in ipairs(units) do
            if n >= u[1] then return string.format("%.2f%s", n / u[1], u[2]) end
        end
        return tostring(math.floor(n))
    end

    function Util.Dist(a, b)
        if not a or not b then return math.huge end
        return (a - b).Magnitude
    end

    function Util.PlayersList(excludeLocal)
        local t = {}
        for _, p in ipairs(Services.Players:GetPlayers()) do
            if not excludeLocal or p ~= LP then t[#t + 1] = p end
        end
        return t
    end

    function Util.ByNameOrId(key)
        key = tostring(key):gsub("^%s+", ""):gsub("%s+$", "")
        if key == "" then return nil end
        for _, p in ipairs(Services.Players:GetPlayers()) do
            if p.Name:lower() == key:lower()
            or p.DisplayName:lower() == key:lower()
            or tostring(p.UserId) == key then
                return p
            end
        end
        return nil
    end

    function Util.Char(p)
        p = p or LP
        local c = p.Character
        if c and c.Parent and c:FindFirstChild("HumanoidRootPart") then return c end
        return nil
    end

    function Util.Root(p)
        local c = Util.Char(p)
        return c and c:FindFirstChild("HumanoidRootPart") or nil
    end

    function Util.Hum(p)
        local c = Util.Char(p)
        return c and c:FindFirstChildOfClass("Humanoid") or nil
    end

    -- pega o objeto do ovo a partir de um prompt (usado no steal)
    function Util.PromptEgg(prompt)
        for _, k in ipairs({ "Egg", "ObjectValue", "Target", "Parent" }) do
            local o = prompt:FindFirstChild(k)
            if o and o:IsA("ObjectValue") and o.Value then return o.Value end
        end
        local p = prompt.Parent
        if p and p.Name ~= "SmartPromptPart" then return p end
        return prompt
    end
end

function UI.New(class, props, parent)
    local i = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then i[k] = v end
    end
    i.Parent = parent
    return i

function UI.Corner(obj, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 6)
    c.Parent = obj
    return c
end

function UI.Stroke(obj, col, thick, trans)
    local s = Instance.new("UIStroke")
    s.Color = col or UI.Theme.Stroke
    s.Thickness = thick or 1
    s.Transparency = trans or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = obj
    return s
end

function UI.Padding(obj, l, r, t, b)
    local p = Instance.new("UIPadding")
    p.PaddingLeft = UDim.new(0, l or 0)
    p.PaddingRight = UDim.new(0, r or 0)
    p.PaddingTop = UDim.new(0, t or 0)
    p.PaddingBottom = UDim.new(0, b or 0)

    l.Padding = UDim.new(0, padding or 6)
    l.SortOrder = Enum.SortOrder.LayoutOrder
    l.FillDirection = dir or Enum.FillDirection.Vertical
    l.Parent = obj
    return l
end

function UI:Tab(name, icon)
    local btn = UI.New("TextButton", {
        Size = UDim2.new(1, 0, 0, 32),
        BackgroundColor3 = UI.Theme.Surface,
        Font = UI.Theme.Font, TextSize = 12,
        TextColor3 = UI.Theme.SubText,
        TextXAlignment = Enum.TextXAlignment.Left,
        Text = "   " .. (icon and (icon .. "  ") or "") .. name,
        AutoButtonColor = false,
    }, self.Rail)
    UI.Corner(btn, 7)

    local page = UI.New("ScrollingFrame", {
        BackgroundTransparency = 1,
        Size = UDim2.new(1, 0, 1, 0),
        CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        ScrollBarThickness = 3,
        ScrollBarImageColor3 = UI.Theme.Stroke,
        BorderSizePixel = 0,
        Visible = false,
    }, self.Content)
    UI.List(page, 8)
    UI.Padding(page, 0, 6, 0, 10)

    local tab = { Button = btn, Page = page }

    local function select()
        for _, t in pairs(self.Tabs or {}) do
            t.Button.BackgroundColor3 = UI.Theme.Surface
            t.Button.TextColor3 = UI.Theme.SubText
            t.Page.Visible = false
        end
        btn.BackgroundColor3 = UI.Theme.Accent
        btn.TextColor3 = Color3.new(1, 1, 1)
        page.Visible = true
    end
    btn.MouseButton1Click:Connect(select)

    self.Tabs = self.Tabs or {}
    table.insert(self.Tabs, tab)
    if #self.Tabs == 1 then select() end
    return tab
end

    local function render()

    local function set(v, fire)
        state = v and true or false
        Flags[key] = state

    end

    return row
end

    local function render()
        local a = (value - min) / math.max(max - min, 1e-9)
        fill.Size = UDim2.new(Util.Clamp(a, 0, 1), 0, 1, 0)
        val.Text = tostring(Util.Round(value, dec))
    end

    local function apply(x)
        local a = Util.Clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
        value = Util.Round(min + (max - min) * a, dec)
        Flags[key] = value
        render()
        if cb then cb(value) end
    end

    bar.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            apply(i.Position.X)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement
        or i.UserInputType == Enum.UserInputType.Touch) then
            apply(i.Position.X)
        end
    end)
    UIS.InputEnded:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.MouseButton1
        or i.UserInputType == Enum.UserInputType.Touch then dragging = false end
    end)

    render()
    UI.Elements[key] = {
        set = function(v) value = tonumber(v) or min; Flags[key] = value; render() end,
        get = function() return value end,
    }
    return row
end

do
    Egg.RarityOrder = {
        "Comum", "Incomum", "Raro", "Épico", "Lendário",
        "Mítico", "Cósmico", "Secreto", "Eterno", "Divino",
    }
    Egg.RarityKeywords = {
        { "divine", "Divino" }, { "eternal", "Eterno" }, { "secret", "Secreto" },
        { "cosmic", "Cósmico" }, { "mythic", "Mítico" }, { "legendary", "Lendário" },
        { "epic", "Épico" }, { "rare", "Raro" }, { "uncommon", "Incomum" },
    }
    Egg.MutationKeywords = {
        { "spiritbloom", "Spirit Bloom", 3.0 }, { "rainbow", "Rainbow", 2.5 },
        { "golden", "Golden", 2.0 }, { "bloom", "Bloom", 1.5 },
        { "silver", "Silver", 1.25 }, { "scrambled", "Scrambled", 1.2 },
        { "fractured", "Fractured", 1.15 }, { "monstrous", "Monstrous", 1.1 },
    }
    Egg.Cache = { eggs = {}, prompts = {}, at = 0 }

    function Egg.Container()
        return WS:FindFirstChild("AreaEggSlotsClient")
    end

    function Egg.PromptPart()
        return WS:FindFirstChild("SmartPromptPart")
    end

    -- marca se o ovo está no nosso plot / na mão de alguém
    local function ownedInfo(obj)
        local plot = WS:FindFirstChild("Plots")
        if plot then
            local anc = obj
            for _ = 1, 6 do
                if not anc or anc == WS then break end
                if anc.Parent == plot then
                    local owner = anc:GetAttribute("Owner") or anc.Name
                    return owner == LP.Name or owner == LP.DisplayName
                end
                anc = anc.Parent
            end
        end
        local anc = obj
        for _ = 1, 5 do
            if not anc or anc == WS then break end
            local hum = anc:FindFirstChildOfClass("Humanoid")
            if hum then
                local plr = Services.Players:GetPlayerFromCharacter(anc)
                if plr then return plr == LP, plr end
            end
            anc = anc.Parent
        end
        return false, nil
    end

    -- extrai nome/raridade/mutação do ovo a partir de várias fontes
    function Egg.ReadInfo(obj)
        local names, attrs = {}, {}
        local function collect(o, depth)
            if not o or depth > 3 then return end
            names[#names + 1] = o.Name
            for k, v in pairs(o:GetAttributes()) do
                if type(v) == "string" then attrs[#attrs + 1] = k .. "=" .. v end
            end
            for _, c in ipairs(o:GetChildren()) do
                if c:IsA("ValueBase") and type(c.Value) == "string" then
                    attrs[#attrs + 1] = c.Value
                end
                if c:IsA("BasePart") or c:IsA("Model") then collect(c, depth + 1) end
            end
        end
        collect(obj, 0)

        local blob = table.concat(names, " ") .. " " .. table.concat(attrs, " "):lower()
        local blobLow = blob:lower()

        local rarity, petName, income, mutation, mult = "Comum", nil, 0, nil, 1

        for _, k in ipairs(Egg.RarityKeywords) do
            if blobLow:find(k[1], 1, true) then rarity = k[2] break end
        end
        for _, k in ipairs(Egg.MutationKeywords) do
            if blobLow:find(k[1], 1, true) then
                mutation, mult = k[2], k[3]
                break
            end
        end

        -- pet: primeiro token com capitalização tipo CamelCase
        for _, n in ipairs(names) do
            local clean = n:gsub("Egg", ""):gsub("_", ""):gsub("-", ""):gsub("%s+", "")
            if clean ~= "" and clean:match("^%u") and #clean > 2 then
                petName = n
                break
            end
        end
        for _, a in ipairs(attrs) do
            local v = a:match("^PetName=(.+)$")
            if v then petName = v break end
            local i = a:match("^Income=(%d+)")
            if i then income = tonumber(i) end
        end

        local isEgg = obj:FindFirstChild("Hitbox") ~= nil
            or obj.Name:lower():find("egg", 1, true) ~= nil
            or obj:FindFirstChild("Plane") ~= nil
            or obj:FindFirstChild("RareAreaEggHighlight") ~= nil
            or obj:FindFirstChild("MonsterParasiteVisual") ~= nil

        local owned, holder = ownedInfo(obj)
        return {
            obj = obj, isEgg = isEgg, rarity = rarity, pet = petName or obj.Name,
            income = income, mutation = mutation, mult = mult,
            owned = owned, holder = holder,
            pos = (obj:IsA("BasePart") and obj.Position)
                or (obj:FindFirstChild("Hitbox") and obj.Hitbox.Position)
                or nil,
        }
    end

    function Egg.Scan(force)
        local now = os.clock()
        if not force and now - Egg.Cache.at < 0.5 then return Egg.Cache.eggs end
        Egg.Cache.at = now

        local out, prompts = {}, {}
        local container = Egg.Container()

        if container then
            for _, slot in ipairs(container:GetChildren()) do
                local target = slot:FindFirstChild("Plane")
                    or slot:FindFirstChild("Hitbox")
                    or (slot:IsA("BasePart") and slot)
                    or slot:FindFirstChildWhichIsA("BasePart")
                if target then
                    local info = Egg.ReadInfo(target)
                    info.slot = slot
                    out[#out + 1] = info
                end
            end
        end

        -- ovos de evento / soltos
        for _, o in ipairs(WS:GetChildren()) do
            if (o:IsA("Model") or o:IsA("BasePart"))
            and (o.Name:lower():find("egg", 1, true) or o:FindFirstChild("Hitbox")) then
                local info = Egg.ReadInfo(o)
                info.pos = info.pos or (o:IsA("BasePart") and o.Position)
                if info.pos then out[#out + 1] = info end
            end
        end

        local pp = Egg.PromptPart()
        if pp then
            for _, p in ipairs(pp:GetChildren()) do
                if p:IsA("ProximityPrompt") and p.Enabled then
                    prompts[#prompts + 1] = p
                end
            end
        end

        Egg.Cache.eggs, Egg.Cache.prompts = out, prompts
        return out
    end

    function Egg.Rank(rarity)
        for i, r in ipairs(Egg.RarityOrder) do
            if r == rarity then return i end
        end
        return 1
    end

    function Egg.Best(minRarity, maxDist)
        local me = Util.Root(LP)
        local min = Egg.Rank(minRarity or "Comum")
        local best, bestScore
        for _, e in ipairs(Egg.Scan()) do
            if e.isEgg and e.pos and not (Flags.Egg_SkipOwnBase and e.owned) then
                local okRar = Egg.Rank(e.rarity) >= min
                local okInc = (e.income or 0) >= (Flags.Egg_IncomeMin or 0)
                local d = me and Util.Dist(me.Position, e.pos) or 0
                if okRar and okInc and d <= (maxDist or Flags.Egg_MaxDist) then
                    local score = (Flags.Egg_MutationPriority and e.mult or 1) * 1000
                        + Egg.Rank(e.rarity) * 100 - d / 100
                    if not bestScore or score > bestScore then
                        best, bestScore = e, score
                    end
                end
            end
        end
        return best
    end

    -- dispara o prompt do ovo
    function Egg.FirePrompt(prompt)
        pcall(function() prompt.HoldDuration = 0 end)
        pcall(function() prompt.MaxActivationDistance = 100 end)
        if type(fireproximityprompt) == "function" then
            local ok = pcall(fireproximityprompt, prompt, 0)
            if ok then return true end
        end
        local ok = pcall(function()
            prompt:InputHoldBegin()
            task.wait(0.05)
            prompt:InputHoldEnd()
        end)
        return ok
    end

    function Egg.PromptsNear(pos, radius)
        local out = {}
        for _, p in ipairs(Egg.Cache.prompts) do
            local a = p.Parent
            if a and a:IsA("BasePart") and Util.Dist(a.Position, pos) <= radius then
                out[#out + 1] = p
            end
        end
        return out
    end
end

do
    State.FlyBV, State.FlyBG = nil, nil
    State.NoClipConn = nil

    function PlayerMod.SetSpeed(v)
        local h = Util.Hum(LP)

    function PlayerMod.Teleport(pos, speed)
        local root = Util.Root(LP)
        if not root or not pos then return false end
        speed = speed or Flags.Egg_StealSpeed or 120
        local dist = Util.Dist(root.Position, pos)
        if dist <= 8 then
            root.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
            return true
        end
        local dir = (pos - root.Position).Unit
        local steps = math.ceil(dist / math.max(speed * 0.1, 4))
        for i = 1, steps do
            if not root or not root.Parent then return false end
            root.CFrame = root.CFrame + dir * (dist / steps)
            Run.Heartbeat:Wait()
        end
        return true
    end

    function PlayerMod.StopFly()
        local root = Util.Root(LP)
        if root then
            for _, n in ipairs({ "VH_FlyBV", "VH_FlyBG" }) do
                local o = root:FindFirstChild(n)
                if o then o:Destroy() end
            end
        end
    end

    function PlayerMod.StartFly()
        local root = Util.Root(LP)
        if not root then return end
        PlayerMod.StopFly()
        local bv = Instance.new("BodyVelocity")
        bv.Name = "VH_FlyBV"
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bv.Velocity = Vector3.zero
        bv.Parent = root
        local bg = Instance.new("BodyGyro")
        bg.Name = "VH_FlyBG"
        bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
        bg.P = 1500
        bg.Parent = root
        State.FlyBV, State.FlyBG = bv, bg
    end

    function PlayerMod.FlyLoop()
        Run.RenderStepped:Connect(function()
            if not Flags.Plr_Fly then return end
            local root, cam = Util.Root(LP), Services.Cam
            if not root or not cam then return end
            local bv, bg = State.FlyBV, State.FlyBG
            if not bv or not bv.Parent then PlayerMod.StartFly() return end
            bg.CFrame = cam.CFrame
            local move = Vector3.zero
            local sp = Flags.Plr_FlySpeed
            if UIS:IsKeyDown(Enum.KeyCode.W) then move += cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.S) then move -= cam.CFrame.LookVector end
            if UIS:IsKeyDown(Enum.KeyCode.A) then move -= cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.D) then move += cam.CFrame.RightVector end
            if UIS:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
            if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then move -= Vector3.new(0, 1, 0) end
            if move.Magnitude > 0 then move = move.Unit * sp end
            bv.Velocity = move
        end)
    end

    function PlayerMod.SetNoClip(on)
        if on and not State.NoClipConn then
            State.NoClipConn = Run.Stepped:Connect(function()
                if not Flags.Plr_Noclip then return end
                local c = Util.Char(LP)
                if not c then return end
                for _, p in ipairs(c:GetDescendants()) do
                    if p:IsA("BasePart") and p.CanCollide then p.CanCollide = false end
                end
            end)
        elseif not on and State.NoClipConn then
            State.NoClipConn:Disconnect()
            State.NoClipConn = nil
        end
    end

    function PlayerMod.Reset()
        local h = Util.Hum(LP)
        if h then h.Health = 0 end
    end
end

do
    Combat.Target = nil
    Combat.LastRemote = nil

    local function visible(part)
        if not Flags.Aim_VisCheck then return true end
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = { LP.Character }
        local res = WS:Raycast(Services.Cam.CFrame.Position,
            (part.Position - Services.Cam.CFrame.Position), params)
        return res and (res.Instance:IsDescendantOf(part.Parent) or res.Instance == part)
    end

    local function boneOf(char)
        local b = Flags.Aim_Bone
        if b == "Nearest" then
            local best, bd
            for _, n in ipairs({ "Head", "HumanoidRootPart", "UpperTorso", "Torso" }) do
                local p = char:FindFirstChild(n)
                if p then
                    local d = Util.Dist(Services.Cam.CFrame.Position, p.Position)
                    if not bd or d < bd then best, bd = p, d end
                end
            end
            return best
        end
        return char:FindFirstChild(b) or char:FindFirstChild("HumanoidRootPart")
    end

    function Combat.InFov(worldPos, radius)
        local cam = Services.Cam
        local sp = cam:WorldToViewportPoint(worldPos)
        if sp.Z <= 0 then return false end
        local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
        local diff = (Vector2.new(sp.X, sp.Y) - center).Magnitude
        return diff <= radius
    end

    function Combat.PickTarget()
        local cam = Services.Cam
        local best, bestScore
        local targets = Flags.Aim_Targets

        local function consider(pos, obj, kind)
            if not pos then return end
            if not Combat.InFov(pos, Flags.Aim_Fov * (cam.ViewportSize.Y / 720)) then return end
            local d = Util.Dist(cam.CFrame.Position, pos)
            if d > Flags.Aim_MaxDist then return end
            local score = -d
            if Flags.Aim_Priority == "Menor vida" and kind == "player" then
                local h = obj.Character and obj.Character:FindFirstChildOfClass("Humanoid")
                if h then score = -h.Health end
            elseif Flags.Aim_Priority == "FOV" then
                local sp = cam:WorldToViewportPoint(pos)
                local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
                score = -(Vector2.new(sp.X, sp.Y) - center).Magnitude
            end
            if not bestScore or score > bestScore then
                best, bestScore = { part = obj, pos = pos, kind = kind }, score
            end
        end

        local includePlayers = false
        local includeEggs = Flags.Aim_LockEgg
        for _, t in ipairs(targets) do
            if t == "Players" then includePlayers = true end
            if t == "Ovos" then includeEggs = true end
        end

        if includePlayers then
            for _, p in ipairs(Util.PlayersList(true)) do
                local char = Util.Char(p)
                if char then
                    if Flags.Aim_TeamCheck and p.Team and LP.Team and p.Team == LP.Team then
                        -- segue
                    else
                        local part = boneOf(char)
                        if part and visible(part) then
                            consider(part.Position, p, "player")
                        end
                    end
                end
            end
        end

        if includeEggs then
            for _, e in ipairs(Egg.Scan()) do
                if e.pos then consider(e.pos, e.obj, "egg") end
            end
        end

        Combat.Target = best
        return best
    end

    function Combat.AimStep()
        if not Flags.Aim_Enabled then return end
        local target = Combat.PickTarget()
        if not target then return end

        local cam = Services.Cam
        local pos = target.pos
        if Flags.Aim_Prediction > 0 and target.kind == "player" then
            local root = Util.Root(target.part)
            if root then pos = pos + root.Velocity * Flags.Aim_Prediction end
        end

        local goal = CFrame.new(cam.CFrame.Position, pos)
        local a = 1 - Flags.Aim_Smooth
        cam.CFrame = cam.CFrame:Lerp(goal, Util.Clamp(a, 0.02, 1))
    end

    -- -------- Silent Aim: hook no __namecall --------
    function Combat.InstallSilent()
        local mt = getrawmetatable(game)
        if not mt then return end
        local old = mt.__namecall
        setreadonly(mt, false)
        mt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            if Flags.Silent_Enabled and (method == "FireServer" or method == "InvokeServer") then
                if Flags.Silent_HitChance >= 100 or math.random(1, 100) <= Flags.Silent_HitChance then
                    local t = Combat.Target or Combat.PickTarget()
                    if t then
                        local args = { ... }
                        local replaced = false
                        for i, v in ipairs(args) do
                            if typeof(v) == "Vector3" then
                                args[i] = t.pos
                                replaced = true
                            elseif typeof(v) == "CFrame" then
                                args[i] = CFrame.new(Services.Cam.CFrame.Position, t.pos)
                                replaced = true
                            end
                        end
                        if replaced then
                            Combat.LastRemote = self
                            if Flags.Silent_Debug then
                                print("[VeniceHub][Silent]", self.Name, method)
                            end
                            return old(self, table.unpack(args))
                        end
                    end
                end
            end
            return old(self, ...)
        end)
        setreadonly(mt, true)
        Notify.Push("Silent Aim", "Hook de rede instalado.", 3, "ok")
    end
end

do
    function Egg.MyBasePos()
        local plots = WS:FindFirstChild("Plots")
        if not plots then return nil end
        for _, pl in ipairs(plots:GetChildren()) do
            local owner = pl:GetAttribute("Owner") or pl.Name
            if owner == LP.Name or owner == LP.DisplayName
            or (LP:GetAttribute("Plot") and LP:GetAttribute("Plot") == pl.Name) then
                local base = pl:FindFirstChildWhichIsA("BasePart")
                    or pl:FindFirstChild("Base")
                if base then return base.Position end
                return pl:GetPivot().Position
            end
        end
        -- fallback: spawn
        local sp = WS:FindFirstChildOfClass("SpawnLocation")
        return sp and sp.Position or nil
    end

    function Egg.TpToEgg(info)
        if not info or not info.pos then return false end
        return PlayerMod.Teleport(info.pos, Flags.Egg_StealSpeed)
    end

    function Egg.Grab(info)
        if not info then return false end
        local target = info.pos
        if not target then return false end

        for _ = 1, Flags.Egg_Retries do
            PlayerMod.Teleport(target, Flags.Egg_StealSpeed)
            task.wait(0.05)

            if Flags.Egg_UsePrompt then
                local pp = Egg.PromptPart()
                if pp then
                    for _, p in ipairs(pp:GetChildren()) do
                        if p:IsA("ProximityPrompt") and p.Enabled then
                            local a = p.Parent
                            if a and a:IsA("BasePart") and Util.Dist(a.Position, target) <= Flags.Egg_ApproachDist + 4 then
                                Egg.FirePrompt(p)
                            end
                        end
                    end
                end
            end

            if Flags.Egg_UseTouch then
                local root = Util.Root(LP)
                local part = info.obj:IsA("BasePart") and info.obj
                    or info.obj:FindFirstChildWhichIsA("BasePart")
                if root and part then
                    firetouchinterest(root, part, 0)
                    task.wait()
                    firetouchinterest(root, part, 1)
                end
            end

            if Flags.Egg_UseRemote and Combat.LastRemote then
                pcall(function()
                    Combat.LastRemote:FireServer("Grab", info.obj)
                end)
            end

            task.wait(Flags.Egg_StealDelay)
            if not Flags.Egg_Verify then return true end

            -- verifica se o ovo saiu do slot
            if info.slot and not info.slot.Parent then return true end
            local c = Util.Char(LP)
            if c then
                for _, d in ipairs(c:GetDescendants()) do
                    if d:IsA("BasePart") and d:FindFirstChild("Hitbox") then return true end
                end
            end
        end
        return false
    end

    function Egg.ReturnBase()
        local base = Egg.MyBasePos()
        if not base then
            Notify.Push("Base", "Base não encontrada.", 3, "warn")
            return false
        end
        PlayerMod.Teleport(base, Flags.Egg_ReturnSpeed)
        if Flags.Egg_AutoDeposit then
            task.wait(0.15)
            local pp = Egg.PromptPart()
            if pp then
                for _, p in ipairs(pp:GetChildren()) do
                    if p:IsA("ProximityPrompt") and p.Enabled then
                        local a = p.Parent
                        if a and a:IsA("BasePart") and Util.Dist(a.Position, base) <= 40 then
                            Egg.FirePrompt(p)
                        end
                    end
                end
            end
        end
        return true
    end

    function Egg.StealLoop()
        task.spawn(function()
            while task.wait(0.2) do
                if not Flags.Egg_AutoSteal then continue end
                local info = Egg.Best(Flags.Egg_RarityMin, Flags.Egg_MaxDist)
                if not info then continue end
                State.CurrentEgg = info
                local ok = Egg.Grab(info)
                if ok then
                    State.Stolen = (State.Stolen or 0) + 1
                    if Flags.Egg_AutoReturn then Egg.ReturnBase() end
                end
            end
        end)
    end
end

        local function scan(cont)
            if not cont then return end
            for _, o in ipairs(cont:GetDescendants()) do
                local attrs = o:GetAttributes()
                local id = attrs.SkinId or attrs.Skin or attrs.PetSkin
                    or (o:FindFirstChild("SkinId") and o.SkinId.Value)
                if id then
                    out[#out + 1] = { pet = o.Name, skin = id, obj = o }
                end
            end
        end
        scan(p.Character)
        scan(p:FindFirstChild("Backpack"))
        scan(p:FindFirstChild("PlayerGui"))
        return out
    end

    function Skin.CopyFrom(target, visibleToAll)
        local p = typeof(target) == "Instance" and target or Util.ByNameOrId(target)
        if not p then
            Notify.Push("Skins", "Jogador não encontrado: " .. tostring(target), 4, "err")
            return
        end
        local list = Skin.ExtractFrom(p)
        if #list == 0 then
            Notify.Push("Skins", "Nenhuma skin detectada em " .. p.Name, 4, "warn")
            return
        end
        Skin.Cache[p.Name] = list

        local applied = 0
        for _, item in ipairs(list) do
            local ok = pcall(function()
                if visibleToAll then
                    local remotes = RS:FindFirstChild("Packages")
                        and RS.Packages:FindFirstChild("Networking")
                    if remotes then
                        for _, r in ipairs(remotes:GetDescendants()) do
                            if r:IsA("RemoteEvent") and r.Name:lower():find("skin", 1, true) then
                                r:FireServer("Equip", item.pet, item.skin)
                            end
                        end
                    end
                end
            end)
            if ok then applied += 1 end
        end
        Notify.Push("Skins", string.format("%d skin(s) copiada(s) de %s%s",
            #list, p.Name, visibleToAll and " (público)" or " (local)"), 4, "ok")
    end

    function Skin.Loop()
        task.spawn(function()
            while task.wait(Flags.Skin_ScanDelay) do
                if Flags.Skin_AutoApply and Flags.Skin_Target ~= "" then
                    Skin.CopyFrom(Flags.Skin_Target, Flags.Skin_VisibleToAll)
                end
            end
        end)
    end
end

--=====================================================================
-- 15) CONFIG: save / load / autosave / import / export
--=====================================================================
do
    Cfg.Blacklist = {
        Sys_UiKey = true, Sys_PanicKey = true,
        Esp_BoxColor = true, Esp_EggColor = true, Fov_Color = true,
    }

    function Cfg.Serialize()
        local out = {}
        for k, v in pairs(Flags) do
            if not Cfg.Blacklist[k] and type(v) ~= "function" and type(v) ~= "userdata" then
                out[k] = v
            end
        end
        return out
    end

    function Cfg.Apply(data)
        local n = 0
        for k, v in pairs(data) do
            if Flags[k] ~= nil and UI.Elements[k] then
                UI.Elements[k].set(v)
                n += 1
            end
        end
        Notify.Push("Config", n .. " opções aplicadas.", 3, "ok")
    end

    function Cfg.Save(slot)
        slot = slot or "default"
        if not Util.FileAPI() then
            Util.Clipboard(Services.Http:JSONEncode(Cfg.Serialize()))
            Notify.Push("Config", "Sem API de arquivo — JSON copiado.", 4, "warn")
            return
        end
        if type(makefolder) == "function" then
            pcall(makefolder, "VeniceHub")
            pcall(makefolder, HUB.Folder)
        end
        local path = HUB.Folder .. "/" .. slot .. ".json"
        pcall(writefile, path, Services.Http:JSONEncode(Cfg.Serialize()))
        Notify.Push("Config", "Salvo em " .. slot, 3, "ok")
    end

    function Cfg.Load(slot)
        slot = slot or "default"
        if not Util.FileAPI() then return end
        local path = HUB.Folder .. "/" .. slot .. ".json"
        if not pcall(isfile, path) or not isfile(path) then
            Notify.Push("Config", "Slot '" .. slot .. "' não existe.", 3, "warn")
            return
        end
        local ok, data = pcall(function()
            return Services.Http:JSONDecode(readfile(path))
        end)
        if ok and data then Cfg.Apply(data) else
            Notify.Push("Config", "Falha ao ler config.", 3, "err")
        end
    end

    function Cfg.Export()
        local json = Services.Http:JSONEncode(Cfg.Serialize())
        Util.Clipboard(json)
        Notify.Push("Config", "Config exportada para a área de transferência.", 4, "ok")
    end

    function Cfg.Import()
        local ok, data = pcall(function()
            return Services.Http:JSONDecode(readfile(HUB.Folder .. "/import.json"))
        end)
        if ok and data then Cfg.Apply(data) end
    end

    function Cfg.AutoSave()
        task.spawn(function()
            while task.wait(Flags.Sys_AutoSaveInterval) do
                if Flags.Sys_AutoSave then Cfg.Save("autosave") end
            end
        end)
    end
end

    local statLabel = UI:Label(i2.Body, "")
    task.spawn(function()
        while task.wait(1) do
            local fps = math.floor(1 / math.max(Run.RenderStepped:Wait(), 1e-4))
            statLabel.Text = string.format(
                "Ovos no mapa: %d\nOvos roubados: %d\nEventos ativos: %d\nFPS: ~%d",
                #Egg.Scan(), State.Stolen or 0, #Events.Active, fps)
        end
    end)

    local i3 = UI:Section(tInfo, "Diagnóstico do jogo")
    UI:Button(i3.Body, "Dump da estrutura (copiar)", function()
        local lines = {}
        local function dump(o, depth)
            if depth > 3 then return end
            lines[#lines + 1] = string.rep("  ", depth) .. o.Name .. " [" .. o.ClassName .. "]"
            for _, c in ipairs(o:GetChildren()) do dump(c, depth + 1) end
        end
        for _, name in ipairs({ "AreaEggSlotsClient", "SmartPromptPart", "Plots" }) do
            local o = WS:FindFirstChild(name)
            if o then dump(o, 0) else lines[#lines + 1] = name .. " -> não encontrado" end
        end
        Util.Clipboard(table.concat(lines, "\n"))
        Notify.Push("Dump", #lines .. " linhas copiadas.", 4, "info")
    end)
    UI:Button(i3.Body, "Checar compatibilidade", function()
        local rep = {}
        rep[#rep + 1] = "AreaEggSlotsClient: " .. tostring(WS:FindFirstChild("AreaEggSlotsClient") ~= nil)
        rep[#rep + 1] = "SmartPromptPart: " .. tostring(WS:FindFirstChild("SmartPromptPart") ~= nil)
        rep[#rep + 1] = "Plots: " .. tostring(WS:FindFirstChild("Plots") ~= nil)
        rep[#rep + 1] = "Packages.Networking: " .. tostring(RS:FindFirstChild("Packages") ~= nil)
        Notify.Push("Compatibilidade", table.concat(rep, " | "), 6, "info")
    end)

    -- panic key
    UIS.InputBegan:Connect(function(i, gp)
        if gp then return end
        if i.KeyCode == Enum.KeyCode[Flags.Sys_PanicKey] then
            for k, v in pairs(Flags) do
                if type(v) == "boolean" and UI.Elements[k] and k:match("^Aim_")
                or k:match("^Silent_") or k:match("^Esp_") or k:match("^Egg_Auto")
                or k:match("^Farm_Enabled") or k:match("^Event_Auto") or k:match("^Plr_Fly") then
                    UI.Elements[k].set(false)
                end
            end
            Notify.Push("Pânico", "Funções desligadas.", 3, "warn")
        end
    end)
end

--=====================================================================
-- 17) INICIALIZAÇÃO
--=====================================================================
local function Init()
    if LP:GetAttribute("VeniceHubLoaded") then
        local old = (gethui and gethui() or LP:WaitForChild("PlayerGui")):FindFirstChild("VeniceHub")
        if old then old:Destroy() end
    end
    LP:SetAttribute("VeniceHubLoaded", true)

    if game.PlaceId ~= HUB.Place then
        print("[Venice Hub] Aviso: PlaceId diferente (" .. game.PlaceId .. "). Alguns recursos podem não funcionar.")
    end

    BuildPanel()
    Visuals.BuildFov()
    Combat.InstallSilent()

    Egg.StealLoop()
    Farm.Loop()
    Farm.Treadmill()
    Events.Loop()
    Skin.Loop()
    Cfg.AutoSave()
    PlayerMod.Hooks()
    PlayerMod.FlyLoop()
    Visuals.Loop()

    Run.RenderStepped:Connect(function()
        local mode = Flags.Aim_Mode
        local key = Enum.KeyCode[Flags.Aim_Key] or Enum.KeyCode.E
        local active = Flags.Aim_Enabled
        if mode == "Hold" then active = active and UIS:IsKeyDown(key) end
        if mode == "Toggle" then active = Flags.Aim_Enabled end
        if active then Combat.AimStep() end
    end)

    task.spawn(function()
        while task.wait(0.1) do
            local skey = Enum.KeyCode[Flags.Silent_Key] or Enum.KeyCode.Q
            local smode = Flags.Silent_Mode
            if smode == "Camera" or smode == "Híbrido" then
                if Flags.Silent_Enabled and UIS:IsKeyDown(skey) then
                    local t = Combat.PickTarget()
                    if t then
                        Services.Cam.CFrame = CFrame.new(Services.Cam.CFrame.Position, t.pos)
                    end
                end
            end
        end
    end)

    Notify.Push("Pronto", "Painel iniciado. Tecla de pânico: " .. Flags.Sys_PanicKey, 5, "ok")
    print("[Venice Hub] v" .. HUB.Version .. " carregado em " .. game.JobId)
end

-- pcall de segurança na inicialização
local ok, err = pcall(Init)
if not ok then
    warn("[Venice Hub] Erro na inicialização: " .. tostring(err))
end

return HUB
