local function new(class, props, children)
    local inst = Instance.new(class)
    if props then
        local parent = props.Parent
        for k, v in pairs(props) do
            if k ~= "Parent" then pcall(function() inst[k] = v end) end
        end
        if parent then inst.Parent = parent end
    end
    if children then for _, c in ipairs(children) do c.Parent = inst end end
    return inst
end

local function corner(inst, r) return new("UICorner", { CornerRadius = UDim.new(0, r or 10), Parent = inst }) end
local function stroke(inst, c, t, tr)
    return new("UIStroke", { Color = c or CONFIG.card, Thickness = t or 1, Transparency = tr or 0.4,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border, Parent = inst })
end
local function grad(inst, c1, c2, rot)
    return new("UIGradient", { Color = ColorSequence.new(c1 or CONFIG.accent, c2 or CONFIG.accent2),
        Rotation = rot or 0, Parent = inst })
end
local function pad(inst, t, r, b, l)
    return new("UIPadding", { PaddingTop = UDim.new(0, t or 0), PaddingRight = UDim.new(0, r or t or 0),
        PaddingBottom = UDim.new(0, b or t or 0), PaddingLeft = UDim.new(0, l or r or t or 0), Parent = inst })
end
local function tween(inst, time, props)
    local t = TweenService:Create(inst, TweenInfo.new(time or 0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    t:Play(); return t
end

local function findPlayer(query)
    if not query or query == "" then return nil end
    local q = query:lower()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LP and (p.Name:lower() == q or p.DisplayName:lower() == q or p.Name:lower():find(q, 1, true)) then

        end

local Bridge = { remotes = {}, funcs = {}, spoofed = {}, scanDone = false }

-- 4.1 Varredura de remotes candidatos a sistema de comando
function Bridge.scan()
    Bridge.remotes, Bridge.funcs = {}, {}
    local keys = { "command", "cmd", "comando", "admin", "painel", "panel", "action",
                   "acao", "ação", "paredao", "paredão", "exec", "cheat", "evento" }
    for _, d in ipairs(ReplicatedStorage:GetDescendants()) do
        if d:IsA("RemoteEvent") then
            local n = d.Name:lower()
            for _, k in ipairs(keys) do
                if n:find(k, 1, true) then table.insert(Bridge.remotes, d); break end
            end
        elseif d:IsA("RemoteFunction") then
            local n = d.Name:lower()
            for _, k in ipairs(keys) do
                if n:find(k, 1, true) then table.insert(Bridge.funcs, d); break end
            end
        end
    end
    Bridge.scanDone = true
    return #Bridge.remotes + #Bridge.funcs
end

-- 4.2 Spoof de permissão client-side (attributes + leaderstats + métodos)
function Bridge.spoof()
    local done = {}
    local attrNames = { "Admin", "IsAdmin", "Owner", "IsOwner", "VIP", "Vip", "Rank",
                        "RankLevel", "PermissionLevel", "Permission", "Staff", "Mod", "Access", "Level" }
    for _, a in ipairs(attrNames) do
        pcall(function()
            local low = a:lower()
            if low:find("level") or low:find("rank") then
                LP:SetAttribute(a, 999)
            else
                LP:SetAttribute(a, true)
            end
            done[#done + 1] = a
        end)
    end

    local ls = LP:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if (v:IsA("IntValue") or v:IsA("NumberValue")) then
                local n = v.Name:lower()
                if n:find("rank") or n:find("level") or n:find("admin") or n:find("perm") then
                    v.Value = 999
                    done[#done + 1] = "leaderstats." .. v.Name
                end
            end
        end
    end

    -- Shadow de métodos usados em checagem client-side
    pcall(function() LP.GetRankInGroup  = function() return 255 end end)
    pcall(function() LP.IsInGroup       = function() return true end end)
    pcall(function() LP.GetRoleInGroup  = function() return "Owner" end end)

    Bridge.spoofed = done
    return #done
end

function Bridge.unspoof()
    pcall(function() LP.GetRankInGroup = nil end)
    pcall(function() LP.IsInGroup = nil end)
    pcall(function() LP.GetRoleInGroup = nil end)
end

-- 4.3 Envio agressivo: remotes + chat (fallback garantido)
function Bridge.chat(msg)
    local ok = pcall(function()
        ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All")
    end)
    if ok then return true end
    return pcall(function()
        TextChatService.ChatInputBarConfiguration.TargetTextChannel:SendAsync(msg)
    end)
end

function Bridge.send(cmdName, args)
    if not Bridge.scanDone then Bridge.scan() end
    local payload = table.concat(args or {}, " ")
    local full = cmdName .. (payload ~= "" and (" " .. payload) or "")
    local hits = 0

    -- a) FireServer em todos os remotes candidatos (formas comuns)
    for _, r in ipairs(Bridge.remotes) do
        pcall(function() r:FireServer(full) end)
        pcall(function() r:FireServer(cmdName, table.unpack(args or {})) end)
        hits = hits + 1
    end

    -- b) Chat com prefixo (sistema de comando padrão)
    for _, pre in ipairs(CONFIG.prefixes) do
        Bridge.chat(pre .. full)
    end

    return hits
end

local function findCommand(name)
    name = name:lower():gsub("^[:/]", "")
    for _, d in ipairs(Commands) do
        if d.name:lower() == name then return d end
        for _, al in ipairs(d.aliases) do if al:lower() == name then return d end end
        for alias in d.name:gmatch("[^|]+") do if alias:lower() == name then return d end end
    end
    return nil
end

local refreshRows -- forward

local function runToggle(def, force)
    local on = force
    if on == nil then on = not def.get() end
    local ok, err = pcall(def.set, on)
    if not ok then
        toast("Erro: " .. tostring(err), "bad")
        log("bad", "✖ " .. def.name .. ": " .. tostring(err))
    else
        log("ok", (on and "✔ " or "✖ ") .. def.name .. (on and " ATIVADO" or " DESATIVADO"))
    end
    if refreshRows then refreshRows() end
end

local function dispatch(raw)
    raw = (raw or ""):gsub("^%s+", ""):gsub("%s+$", "")
    if raw == "" then return false end

    local name, rest = raw:match("^(%S+)%s*(.*)$")
    rest = rest or ""
    local args = {}
    for w in rest:gmatch("%S+") do table.insert(args, w) end

    local def = findCommand(name)
    if not def then return false end

    if def.type == "toggle" then
        local mode = args[1] and args[1]:lower()
        runToggle(def, (mode == "on") and true or (mode == "off") and false or nil)
        return true
    end

    local ok, err = pcall(def.run, args, rest, raw)
    if ok then
        log("ok", "✔ " .. raw)
    else
        log("bad", "✖ " .. tostring(err))
        toast("Erro: " .. tostring(err), "bad")
    end
    return true
end

local function sendCommand()
    local txt = cmdInput.Text
    if txt == "" then return end
    playClick()
    btnSend.Text = "⏳ ..."

    local ok = dispatch(txt)
    if not ok then
        local hits = Bridge.send(txt, {})
        log("info", "↗ enviado ao servidor (" .. hits .. " canais)")
        toast("Comando repassado ao servidor via bridge", "info")
    end

    task.delay(0.6, function() btnSend.Text = "▶ ENVIAR" end)
end

btnSend.MouseButton1Click:Connect(sendCommand)
cmdInput.FocusLost:Connect(function(enter) if enter then sendCommand() end end)

table.sort(Commands, function(a, b)
    local ia, ib = 99, 99
    for i, c in ipairs(CAT_ORDER) do
        if a.cat == c then ia = i end
        if b.cat == c then ib = i end
    end
    if ia ~= ib then return ia < ib end
    if a.type ~= b.type then return a.type < b.type end
    return a.name < b.name
end)

local function applyFilter()
    local q = searchInput.Text:lower()
    local catVisible = {}
    for _, d in ipairs(Commands) do
        local r = Rows[d.name]
        if r then

            if vis then catVisible[d.cat] = true end

end

local function makeDraggable(target, handle)
    handle = handle or target
    local dragging, dragStart, startPos = false, nil, nil

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging, dragStart, startPos = true, input.Position, target.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                        startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

local function clampToScreen(frame, w, h)
    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
    frame.Position = UDim2.new(0, math.clamp(frame.Position.X.Offset, 0, math.max(0, vp.X - w)),
                                0, math.clamp(frame.Position.Y.Offset, 0, math.max(0, vp.Y - h)))
end

local function applyResponsive()
    local cam = workspace.CurrentCamera
    local vp = cam and cam.ViewportSize or Vector2.new(1280, 720)
    local sw = (vp.X * 0.94) / CONFIG.panelSize.X
    local sh = (vp.Y * 0.88) / CONFIG.panelSize.Y
    local scale = math.clamp(math.min(sw, sh), 0.5, 1.25)
    wrapperScale.Scale = scale
    floatScale.Scale = math.clamp(scale, 0.85, 1.1)
    clampToScreen(wrapper, CONFIG.panelSize.X * scale, CONFIG.panelSize.Y * scale)
    clampToScreen(floatBtn, CONFIG.floatSize * floatScale.Scale, CONFIG.floatSize * floatScale.Scale)
end

local function minimize()
    playClick()

    task.delay(0.16, function()
        wrapper.Visible = false
        floatBtn.Visible = true

        tween(floatBtn, 0.18, { Size = UDim2.fromOffset(CONFIG.floatSize, CONFIG.floatSize) })
    end)
    toast("Minimizado — toque no Δ para voltar", "info")
end

    end
end
